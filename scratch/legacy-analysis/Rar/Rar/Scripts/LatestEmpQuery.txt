
select DISTINCT E.EMPLOYEECODE,E.EMPLOYEENAME,NULL,A.AADHARNO,P.MOBILENOPERS,P.MOBILENOOFFC,C.DEPENDTNAME,DV.DIVNFULLNAME,
D.DESGFULLNAME,P.EMAILIDOFFC,B.BASICPAY,B.DATEOFBIRTH,B.SUPERANNUATNDT,NULL,NULL,NULL,NULL,'SC00001',getdate(),E.SERVICESTATCODE from TBAD_EMPLOYEE E 
left join TBAD_DESIGNATIONS D on D.DESGCODE=E.DESGCODE
left join TBAD_DIVISION DV ON DV.DIVNCODE=E.DIVNCODE
left join TBAD_AADHARDETAILS A ON A.EMPLOYEECODE=E.EMPLOYEECODE
left join TBAD_EMPPHONENUMBER P ON P.EMPLOYEECODE=E.EMPLOYEECODE
left join TBAD_DEPENDENT C ON C.EMPLOYEECODE=E.EMPLOYEECODE AND C.SUBCODE='01' 
left join TBAD_BIODATA B ON B.EMPLOYEECODE=E.EMPLOYEECODE
order by E.EMPLOYEENAME

/* Latest Employee Query according to speccs emp table format */
select DISTINCT E.EMPLOYEECODE AS MemEmpCode,E.EMPLOYEENAME AS MemName,NULL AS PanNo,Null AS AadharNo,P.MOBILENOPERS AS Phone,
P.MOBILENOOFFC AS OffPhone,C.DEPENDTNAME AS CareOf,DV.DIVNFULLNAME AS Division,
D.DESGFULLNAME AS Designation,P.EMAILIDOFFC AS MailId,B.BASICPAY AS BasicPay,B.DATEOFBIRTH AS Dob,B.SUPERANNUATNDT AS RetiredDate,
NULL AS BankAccNo,NULL AS IfscCode,NULL AS BankName,NULL AS BankPlace,'SC00001' AS UserId,getdate() AS RegTime,E.SERVICESTATCODE AS Status from TBAD_EMPLOYEE E 
left join TBAD_DESIGNATIONS D on D.DESGCODE=E.DESGCODE
left join TBAD_DIVISION DV ON DV.DIVNCODE=E.DIVNCODE
left join TBAD_AADHARDETAILS A ON A.EMPLOYEECODE=E.EMPLOYEECODE
left join TBAD_EMPPHONENUMBER P ON P.EMPLOYEECODE=E.EMPLOYEECODE
left join TBAD_DEPENDENT C ON C.EMPLOYEECODE=E.EMPLOYEECODE AND C.SUBCODE='01' 
left join TBAD_BIODATA B ON B.EMPLOYEECODE=E.EMPLOYEECODE
order by E.EMPLOYEENAME
